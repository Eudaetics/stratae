    try:
        __slots__ = __var__.get()
        __cache__ = __slots__[4]
    except LookupError:

        __cache__ = __slots__[4]
    if __cache__ is __UNSET__:
        __cache__ = {}
        __slots__[4] = __cache__
    __ck__ = a
