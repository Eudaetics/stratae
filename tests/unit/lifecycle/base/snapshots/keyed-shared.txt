def wrapper(a, b, *args, c, d, **kw):
    __slots__ = __var__.storage
    try:
        __cache__ = __slots__[4]
    except (TypeError, LookupError):
        __slots__ = __scope__.get_slots()
        __cache__ = __slots__[4]
    __ck__ = (a, b, *args, c, d, frozenset(kw.items()), )
    if __cache__ is not __UNSET__:
        try:
            return __cache__[__ck__]
        except KeyError:
            pass
    while True:
        __acquire__()
        try:
            __cache__ = __slots__[4]
            if __cache__ is __UNSET__:
                __cache__ = {}
                __slots__[4] = __cache__
            if __ck__ in __cache__:
                return __cache__[__ck__]
            __pending__ = __inflight__.get(__ck__)
            if __pending__ is None:
                __inflight__[__ck__] = __claimant__()
                break
            __done__ = __watch__(__inflight__, __ck__, __pending__)
        finally:
            __release__()
        __done__.wait()
        if not __slots__:
            __ended__(__scope__)
    try:
        __value__ = __func__(a, b, *args, c=c, d=d, **kw)
        __cache__[__ck__] = __value__
    finally:
        __acquire__()
        try:
            __claim__ = __inflight__.pop(__ck__)
            if __claim__.__class__ is __Pending__:
                __claim__.release()
        finally:
            __release__()
    return __value__
